function doGet(e) { if (e.parameter && e.parameter.action === "saveBattery") { return saveBattery(e.parameter); } if (e.parameter && e.parameter.action === "saveReflection") { return saveReflection(e.parameter); } return getAverage(e); } function getAverage(e) { try { var ss = SpreadsheetApp.openById('106kBkJmQwJ480_d05eoEXIk19qpuWl5mNN64neV7jvU'); var sheet = ss.getSheetByName('Battery'); var data = sheet.getDataRange().getValues(); var batteries = []; for (var i = 1; i < data.length; i++) { var raw = String(data[i][1]).replace('%', '').trim(); var val = parseFloat(raw); if (!isNaN(val) && val > 0) { if (val <= 1) val = Math.round(val * 100); else val = Math.round(val); batteries.push(val); } } var avg = batteries.length > 0 ? Math.round(batteries.reduce(function(a,b){return a+b;},0) / batteries.length) : null; var json = JSON.stringify({ result: 'success', avg: avg, total: batteries.length }); var callback = e.parameter && e.parameter.callback; var output = callback ? callback + "(" + json + ")" : json; return ContentService.createTextOutput(output) .setMimeType(callback ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({ result: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function saveBattery(params) { try { var ss = SpreadsheetApp.openById('106kBkJmQwJ480_d05eoEXIk19qpuWl5mNN64neV7jvU'); var sheet = ss.getSheetByName('Battery'); sheet.appendRow([new Date(), params.battery]); return ContentService.createTextOutput(JSON.stringify({ result: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({ result: 'error', message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function saveReflection(params) { try { var ss = SpreadsheetApp.openById('106kBkJmQwJ480_d05eoEXIk19qpuWl5mNN64neV7jvU'); var sheet = ss.getSheetByName('Reflections'); sheet.appendRow([new Date(), params.takeaway]); return ContentService.createTextOutput(JSON.stringify({ result: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({ result: 'error', message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
